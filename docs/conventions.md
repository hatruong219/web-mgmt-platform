## 1. TypeScript

```typescript
// ✅ DO: Strong types, no any
interface Article {
  id: string
  siteId: string
  title: string
  status: 'draft' | 'published' | 'archived'
}

// ✅ DO: Zod for runtime validation
const createArticleSchema = z.object({
  title: z.string().min(1, 'Title is required'),
  siteId: z.string().uuid(),
})

// ❌ DON'T: any, or loose types
const data: any = await fetch(...)
const article = result as Article
```

### Type Sources
- Database types → `types/database.ts` (generated by Supabase CLI)
- Form types → inferred from Zod schemas (`z.infer<typeof schema>`)
- Component props → defined inline or in same file as component

| Type | Convention | Example |

| Next.js pages | `page.tsx` (fixed) | `app/(dashboard)/page.tsx` |
| Next.js layouts | `layout.tsx` (fixed) | `app/(dashboard)/layout.tsx` |

| Types | `camelCase.ts` | `database.ts` |

## 4. Server Actions

```typescript
// File: lib/actions/articles.ts
'use server'

import { revalidatePath } from 'next/cache'
import { createClient } from '@/lib/supabase/server'
import { createArticleSchema } from '@/lib/validations/article'

export async function createArticleAction(formData: FormData) {
  // 1. Authenticate
  const supabase = await createClient()
  const { data: { user } } = await supabase.auth.getUser()
  if (!user) throw new Error('Unauthorized')

  // 2. Validate input
  const parsed = createArticleSchema.safeParse({
    siteId: formData.get('siteId'),
    title: formData.get('title'),
  })
  if (!parsed.success) return { error: parsed.error.flatten() }

  // 3. Execute
  const { data, error } = await supabase
    .from('articles')
    .insert(parsed.data)
    .select()
    .single()

  if (error) return { error: error.message }

  // 4. Revalidate affected paths
  revalidatePath(`/sites/${parsed.data.siteId}/articles`)

  return { data }
}
```

## 7. Error Handling

```typescript
// ✅ In Server Actions — return error object, don't throw
export async function updateArticle(id: string, data: unknown) {
  try {
    const { error } = await supabase.from('articles').update(data).eq('id', id)
    if (error) return { error: error.message }
    return { success: true }
  } catch (e) {
    return { error: 'Unexpected error occurred' }
  }
}

// ✅ In Server Components — throw to error boundary (or handle gracefully)
const { data, error } = await supabase.from('sites').select('*')
if (error) {
  // Option 1: render error state
  return <ErrorMessage message={error.message} />
  // Option 2: throw (caught by error.tsx boundary)
  throw new Error(error.message)
}
```
